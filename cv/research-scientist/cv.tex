% --- Layout for machine readability ------------------------------------------
% Awesome-CV sets every entry as a two-column table (title | location over
% position | date) with the position line in small caps, and the honors and
% skills sections as three- and two-column tables. Resume parsers read that
% badly, and it was reproduced with pdftotext, pdfminer, pypdf and pdf.js:
%   * the right-hand cells come out as a separate column (pdfminer) or as
%     blocks split by blank lines (pdftotext), so dates detach from entries;
%   * small caps extract as ALL CAPS, and a short ALL-CAPS line is what most
%     parsers take for a section heading, so each job's position line ended
%     the Experience section (the second page of jobs was lost) and the
%     Education entries, "headed" by CORNELL UNIVERSITY, were classified by
%     content, i.e. as jobs;
%   * the footer (name, "Curriculum Vitae", date) sat between the two pages
%     of Experience and was read as a heading or as a job date.
% Everything below avoids those: entries are plain text lines (the location
% and date are pushed right with \hfill on the same line, not a table cell),
% entry lines are mixed case, only section headings extract as ALL CAPS, and
% the footer is a bare page number. Experience must not start right after a
% page break mid-entry; entries are unbreakable blocks. Word spacing is
% widened a little because pypdf (used by Python screeners) drops spaces
% narrower than about a quarter em and ran whole sentences together.
\makeatletter
\newcommand{\acvWordSpace}{\spaceskip=0.28em plus 0.15em minus 0.05em\relax}
\AddToHook{selectfont}{\acvWordSpace}
\emergencystretch=1em

% Section headings in small caps: they read as title case on the page and
% extract as ALL CAPS, which is how parsers recognize a heading.
\renewcommand*{\sectionstyle}[1]{{\fontsize{15pt}{1em}\bodyfont\bfseries\scshape\color{text}#1}}
% Sub-headings (Publications): bold mixed case, so they do not look like headings.
\renewcommand*{\subsectionstyle}[1]{{\fontsize{10.5pt}{1em}\bodyfont\bfseries\color{text} #1}}
% Headline under the name: mixed case instead of small caps.
\renewcommand*{\headerpositionstyle}[1]{{\fontsize{8.5pt}{1em}\bodyfont\color{awesome} #1}}
% Position / institution line of an entry: italic mixed case, not small caps
% (small caps extract as ALL CAPS and were read as section headings).
\renewcommand*{\entrypositionstyle}[1]{{\fontsize{9pt}{1em}\bodyfont\itshape\color{graytext} #1}}
\makeatother

% Separator inside an entry line. Small gaps on purpose: pdfminer starts a
% new text line at a gap wider than about two character widths.
\newcommand{\acvEntrySep}{{\fontsize{9pt}{1em}\selectfont\hspace{0.3em}\textperiodcentered\hspace{0.3em}}}
% Gap between entries. Parsers that split entries on vertical whitespace need
% it clearly larger than the line spacing inside an entry.
\newcommand{\acvEntrySkip}{2.6mm}

% An entry is one unbreakable block of plain text lines (no table):
%   Title (bold) ............................ Location
%   Position (italic) ............................ Date
%   description
% The right-hand field sits on the same text line as the left one, so every
% extractor keeps them together; a table cell would be read as a column.
\newif\ifacvinline % set by \acvinlinetrue inside a section (see cv/education.tex)
\renewenvironment{cventries}{\par\vspace{\acvSectionContentTopSkip}\vspace{-\acvEntrySkip}}{\par}
\renewcommand*{\cventry}[5]{%
  \par\vspace{\acvEntrySkip}%
  \noindent\begin{minipage}{\textwidth}
    \ifacvinline
      % Inline variant (Education): nothing right-aligned. When Education comes
      % before Experience, pdfminer moved the right-aligned school dates below
      % the Experience heading and counted them as jobs.
      \entrytitlestyle{#2}\par\vspace{0.3mm}%
      \entrypositionstyle{#1}\acvEntrySep\entrypositionstyle{#3}\acvEntrySep\entrypositionstyle{#4}\par
    \else
    \ifempty{#2#3}{}{%
      \entrytitlestyle{#2}\hfill\entrylocationstyle{#3}\endgraf% \par is not allowed inside \ifempty
      \vspace{0.3mm}}%
    \entrypositionstyle{#1}\hfill\entrydatestyle{#4}\par
    \fi
    \ifempty{#5}{}{\vspace{0.5mm}\descriptionstyle{#5}}%
  \end{minipage}\par
}

% Bullets inside an entry description. (No ragged2e justify environment: it
% resets \spaceskip, which brought pypdf's run-together words back.)
\renewenvironment{cvitems}{%
  \begin{itemize}[leftmargin=2ex, nosep, noitemsep]
    \setlength{\parskip}{0pt}\acvWordSpace
    \renewcommand{\labelitemi}{\bullet}
}{%
  \end{itemize}
}

% Title-only entry (Research Interests): bold title line, then description.
\newcommand*{\cvtitleentry}[2]{%
  \par\vspace{\acvEntrySkip}%
  \noindent\begin{minipage}{\textwidth}
    \entrytitlestyle{#1}\par
    \vspace{0.4mm}\descriptionstyle{#2}%
  \end{minipage}\par
}

% Honors / service: one line per item, no table. The three-column table was
% read column by column (all years, then all titles, then all places).
\renewenvironment{cvhonors}{\par\vspace{\acvSectionContentTopSkip}\vspace{-1.5mm}\begin{flushleft}\setlength{\parskip}{0.8mm}}{\end{flushleft}}
\renewcommand*{\cvhonor}[4]{%
  \noindent\honorpositionstyle{#1}, \honortitlestyle{#2}\ifempty{#3}{}{\acvEntrySep\honorlocationstyle{#3}}\hfill\honordatestyle{#4}\par
}

% Skills: "Category: items" lines, no table.
\renewenvironment{cvskills}{\par\vspace{\acvSectionContentTopSkip}\vspace{-1.5mm}\begin{flushleft}\setlength{\parskip}{0.6mm}}{\end{flushleft}}
\renewcommand*{\cvskill}[2]{\noindent\skilltypestyle{#1}:\enskip\skillsetstyle{#2}\par}

% Numbered publications list. Inherits \descriptionstyle font (9pt light)

}{%
  \end{enumerate}
}

% Bulleted list in descriptionstyle font (Highlights, Service).
\newenvironment{cvbullets}{%
  \vspace{1mm}%

  \begin{itemize}[leftmargin=2ex, nosep, noitemsep, label=\textbullet]
    \setlength{\parskip}{0pt}
}{%
  \end{itemize}
}
