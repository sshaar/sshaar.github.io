\cvsection{About Me}

\begin{cvparagraph}
I am a PhD candidate at Cornell University, advised by Prof.\ Claire Cardie, working on \textit{Long-Form Input Understanding across Vision-and-Language Tasks}. I have worked across several domains and modalities: document-level event extraction in text (TACL 2025), and open-ended question answering over movie recaps with \textit{MovieRecapsQA} (CVPR 2026) and now over full-length films. During my PhD, I have also collaborated with NewYork-Presbyterian Hospital on cardiology NLP projects, including an LLM-based analysis of UNOS exception requests (JHLT 2026).
Before Cornell, I spent two years as a research assistant at the Qatar Computing Research Institute working on fact-checking and propaganda detection across modalities. I did my undergrad in Computer Science at Carnegie Mellon University.
\end{cvparagraph}

\vspace{-3mm}
\begin{cventries}
  \cvtitleentry
    {Research Interests}
    {
      \begin{cvitems}
        \item \textbf{Long-form input understanding.} How models process long documents, short videos, and full-length films, and what each task requires them to locate in the input.
        \item \textbf{Benchmarks and evaluation for open-ended generation.} Building datasets on which current models fail, and scoring long, open-ended answers against rubrics rather than a single reference answer.
        \item \textbf{Clinical NLP.} Using LLMs to identify clinical decision patterns in unstructured medical narratives, primarily in heart failure and heart transplant care.
        \item \textbf{Fact-checking and propaganda detection} \textit{(prior work).} Detecting previously fact-checked claims, and detecting propaganda and persuasion techniques in text, images, and memes.
      \end{cvitems}
    }
\end{cventries}
