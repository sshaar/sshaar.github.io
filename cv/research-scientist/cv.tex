% --- Spacing fixes ---
% Awesome-CV's cvitems env ends with \vspace{-4mm} which pulls the next
% section heading on top of the last bullet. Redefine without the trailing
% negative space, and give cvsection more breathing room.
\renewenvironment{cvitems}{%
  \vspace{-4.0mm}
  \begin{justify}
  \begin{itemize}[leftmargin=2ex, nosep, noitemsep]
    \setlength{\parskip}{0pt}
    \renewcommand{\labelitemi}{\bullet}
}{%
  \end{itemize}
  \end{justify}
}
\renewcommand{\acvSectionTopSkip}{6mm}

% Title-only cventry variant: a single bold title row + bullet description,
% with no blank position row underneath. Used for Highlights / Research
% Interests where there's no org/location/date metadata.
% Usage: \cvtitleentry{<title>}{<description>}
\newcommand*{\cvtitleentry}[2]{%
  \vspace{-2.0mm}
  \setlength\tabcolsep{0pt}
  \setlength{\extrarowheight}{0pt}
  \begin{tabular*}{\textwidth}{@{\extracolsep{\fill}} L{\textwidth} R{0pt}}
    \entrytitlestyle{#1} & \\
    \multicolumn{2}{L{\textwidth}}{\descriptionstyle{#2}}
  \end{tabular*}%
}

% Numbered publications list. Inherits \descriptionstyle font (9pt light)
% so entries match the rest of the CV's bullets and descriptions.
\newenvironment{publist}{%
  \vspace{1mm}%
  \fontsize{9pt}{1.15em}\bodyfontlight\upshape\color{text}\selectfont
  \begin{enumerate}[leftmargin=2.4em, nosep, itemsep=0.45em, label=\arabic*.]
}{%
  \end{enumerate}
}

% Bulleted list, descriptionstyle font, spacing matched to cvitems
% (used inside cventry descriptions) so a stand-alone Highlights section
% reads identically to bullets inside Research Interests.
\newenvironment{cvbullets}{%
  \vspace{1mm}%
  \fontsize{9pt}{1.15em}\bodyfontlight\upshape\color{text}\selectfont
  \begin{itemize}[leftmargin=2ex, nosep, noitemsep, label=\textbullet]
    \setlength{\parskip}{0pt}
}{%
  \end{itemize}
}
