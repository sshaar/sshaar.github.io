\cvsection{Research Projects}

\newcommand{\cornellnlp}{\href{https://www.cs.cornell.edu/~cardie/}{Cardie Group, Cornell NLP}}

\begin{cventries}

  \cventry
    {Long-form Video QA \& Narrative Understanding}
    {Multimodal benchmarks and models for narrative video question answering}
    {\cornellnlp}
    {2024 -- Present}
    {
      \begin{cvitems}
        \item Designed \textit{MovieRecapsQA}, an open-ended multimodal video QA benchmark grounded in movie recaps (CVPR 2026, first author).
        \item Investigates how vision-language models reason about long narrative video content.
        \item Keywords: Multimodal LLMs, long-form video QA, narrative understanding.
      \end{cvitems}
    }

  \cventry
    {Long-form Text Generation \& Evaluation}
    {Narrative summarization of novels and movies; coherence evaluation}
    {\cornellnlp}
    {2023 -- Present}
    {
      \begin{cvitems}
        \item Building datasets and models for narrative long-document summarization.
        \item Studying evaluation methods that capture semantic coherence across extended passages.
        \item Keywords: Long-document summarization, narrative coherence, evaluation.
      \end{cvitems}
    }

  \cventry
    {Document-Level Event Extraction}
    {Are triggers needed for document-level event extraction?}
    {\cornellnlp}
    {2023 -- 2025}
    {
      \begin{cvitems}
        \item Studied the role of triggers in document-level event-extraction models; published in TACL 2025.
        \item Keywords: Event extraction, document-level IE, structured prediction.
      \end{cvitems}
    }

  \cventry
    {Automated Fact-Checking \& Claim Retrieval (Previous)}
    {Detecting previously fact-checked claims; assisting human fact-checkers}
    {\qcri}
    {2019 -- 2022}
    {
      \begin{cvitems}
        \item Introduced the task of detecting previously fact-checked claims (ACL 2020) and extended it to document-level claim retrieval (Findings of EMNLP 2022).
        \item Co-organized CLEF--CheckThat! Lab tasks on check-worthiness and verified-claim retrieval.
        \item Survey on automated fact-checking for assisting human fact-checkers (IJCAI 2021).
        \item Keywords: Fact-checking, retrieval, misinformation, NLP4SocialGood.
      \end{cvitems}
    }

  \cventry
    {Propaganda \& Disinformation Detection (Previous)}
    {Detecting propaganda techniques in text and memes; COVID-19 infodemic analysis}
    {\qcri}
    {2019 -- 2022}
    {
      \begin{cvitems}
        \item Built \textit{Prta}, a public system for analyzing propaganda techniques in news (ACL 2020 demo, Honorable Mention).
        \item Co-organized SemEval-2021 Task 6 on detecting persuasion techniques in memes.
        \item Multi-stakeholder analysis of the COVID-19 infodemic (ICWSM 2021; Findings of EMNLP 2021).
        \item Keywords: Propaganda detection, multimodal disinformation, COVID-19 infodemic.
      \end{cvitems}
    }

\end{cventries}
